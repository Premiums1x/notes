#隐内  
select e.name,e.age,e.job,d.name from emp e ,dept d where e.dept_id = d.id;  
  
#显内：  
select e.name,e.age,e.job,d.name from emp e inner join dept d on e.dept_id = d.id where e.age < 30;  
  
#有员工的部门、部门id（求部门表和员工表交集部分）  
select distinct dept.id,dept.name from  dept,emp where emp.dept_id = dept.id and emp.dept_id is not null ;  
  
#左外连  
select e.*,d.name from emp e left outer join dept d on e.dept_id = d.id where e.age  > 40;  
  
#不通过外键而通过字段间关系，给两张表创立连接关系：  
select e.*,s.grade from emp e,salgrade s where e.salary between s.losal and  s.hisal;  
  
  
#n张表，有n-1个连接条件：  
select *  
from emp e,  
     dept d,  
     salgrade s  
where (e.dept_id = d.id and e.salary between s.losal and s.hisal)  
  and d.name = '研发部';  
  
#研发部平均工资，两张表：  
select avg(e.salary) from emp e,dept d where e.dept_id = d.id and d.name = '研发部';  
  
#子查询，条件的值来自另一条查询  
select emp.* from emp  where salary > (select salary from emp where name = '灭绝');  

#查工资大于全体员工平均工资的所有员工信息  
select * from emp where salary > (select avg(salary) from emp);  
  


  
select avg(emp.salary) from emp where dept_id = 1;  
  
#低于本部门平均工资的员工信息  
#查的部门就是外层传给内层的（内层根据部门编号查avg得到每个部门avg，外层传入部门编号）  
select * from emp e1 where e1.salary < (select avg(e0.salary) from emp e0 where e0.dept_id = e1.dept_id  );  


#所有部门信息，并统计各部门员工数量  
select d.id,d.name,(select count(*) from emp e where e.dept_id = d.id) '人数' from dept d;  
  
  
#查询所有学生选课情况，展示学生名称、学号、课程名称  
#三张表：student,course,student_course,其中学生和课程是多对多关系，而通过student_course中间表建立主外键关系  
#多表联查关键在于清楚两表间连接关系，消除多余笛卡儿积：  
select student.name,student.no,course.name from student,course,student_course where student.id = student_course.studentid and course.id  = student_course.courseid ;
